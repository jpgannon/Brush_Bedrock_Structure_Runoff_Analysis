recent_data <- read.csv("data/FieldChemData.csv") %>%
  mutate(
    Date = as.Date(Date, format = "%m/%d/%Y"),
    Mountain_Side = case_when(
      Site %in% scarp_sites ~ "Scarp",
      Site %in% dip_sites   ~ "Dip",
      TRUE ~ NA_character_
    ),
    Period = "Recent"
  )

  !names(recent_data) %in% c("Site", "Date", "Mountain_Side", "Period")
]

recent_data <- recent_data %>%
  mutate(across(
    all_of(recent_element_cols),
    ~ as.numeric(gsub(",", "", as.character(.x)))
  ))

ratio_data <- plot_data %>%
  mutate(
    Mg_Ca = if_else(is.finite(Mg) & is.finite(Ca) & Ca > 0, Mg / Ca, NA_real_),
    Na_Ca = if_else(is.finite(Na) & is.finite(Ca) & Ca > 0, Na / Ca, NA_real_)
  )

ratio_df <- ratio_data %>%
  select(Mountain_Side, Period, Side_Period, Mg_Ca, Na_Ca) %>%
  pivot_longer(
    cols = c(Mg_Ca, Na_Ca),
    names_to = "Ratio",
    values_to = "Ratio_Value"
  ) %>%
  mutate(
    Ratio = recode(Ratio,
      Mg_Ca = "Mg / Ca",
      Na_Ca = "Na / Ca"

  ) %>%
  filter(!is.na(Ratio_Value), !is.na(Mountain_Side))

## Delimiter: ","
## dbl  (4): Q_mcd, Qmmd_mcd, Q_beast, Qmmd_beast

## ℹ Use `spec()` to retrieve the full column specification for this data.
## ℹ Specify the column types or set `show_col_types = FALSE` to quiet this message.
# first, make sure datetime is actually parsed correctly
Qdat <- Qdat %>%
  mutate(datetime = parse_date_time(datetime,
                                    orders = c("mdy HMS", "mdy HM", "mdy",
                                               "ymd HMS", "ymd HM", "ymd")))

discharge_long <- Qdat %>%
  select(datetime, Qmmd_mcd, Qmmd_beast) %>%
  pivot_longer(
    cols = c(Qmmd_mcd, Qmmd_beast),
    names_to = "Mountain_Side",
    values_to = "Discharge"
  ) %>%
  mutate(
    Mountain_Side = recode(
      Mountain_Side,
      Qmmd_beast = "Scarp",
      Qmmd_mcd   = "Dip"
    ),
    Window = case_when(
      datetime >= as.POSIXct("2020-08-01") & datetime < as.POSIXct("2020-09-01") ~ "Early August 2020",
      datetime >= as.POSIXct("2020-10-25") & datetime < as.POSIXct("2020-12-01") ~ "Late Fall 2020",
      TRUE ~ NA_character_
    )
  ) %>%
  filter(!is.na(Window)) %>%
  mutate(
    Window = factor(Window, levels = c("Early August 2020", "Late Fall 2020")),
    Mountain_Side = factor(Mountain_Side, levels = c("Scarp", "Dip"))
  )

fig_discharge <- ggplot(
  discharge_long,
  aes(x = datetime, y = Discharge, color = Mountain_Side)
) +
  geom_line(linewidth = 1) +
  facet_wrap(~ Window, ncol = 1, scales = "free_x") +
  scale_color_manual(
    values = c(
      "Scarp" = "#6B5A8E",
      "Dip"   = "#C76A2A"
    )
  ) +
  labs(
    title = "Discharge Comparison",
    x = "",
    y = "Discharge (mm/day)"
  ) +
  poster_theme +
  theme(
    strip.text = element_text(face = "bold", size = 18),
    panel.grid.major.x = element_blank(),
    panel.grid.minor.x = element_blank(),

  )
