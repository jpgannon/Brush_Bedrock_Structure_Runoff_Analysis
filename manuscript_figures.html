water_data <- read.csv("data/FieldChemData.csv")

scarp_sites <- c("FSV1", "BST1", "MPL1", "SRT1", "JPY1", "LSR1")
dip_sites   <- c("MCD1", "SCH1", "CHY1", "DFT1")

water_data <- water_data |> 
  mutate(
    Mountain_Side = case_when(
      Site %in% scarp_sites ~ "Scarp",
      Site %in% dip_sites   ~ "Dip",
      TRUE ~ NA_character_
    ),
    Mountain_Side = factor(
      Mountain_Side,
      levels = c("Scarp", "Dip")
    ),
    Date = as.Date(Date, format = "%m/%d/%Y")
  )

element_cols <- names(water_data)[
  !names(water_data) %in% c("Site", "Date", "Mountain_Side")
]

water_data <- water_data |> 
  mutate(across(
    all_of(element_cols),
    ~ as.numeric(gsub(",", "", as.character(.x)))
  ))

ratio_df <- ratio_data |>
  select(Mountain_Side, Mg_Ca, Na_Ca) |>
  pivot_longer(
    cols = c(Mg_Ca, Na_Ca),
    names_to = "Ratio",
    values_to = "Ratio_Value"
  ) |>
  mutate(
    Ratio = recode(Ratio,
      Mg_Ca = "Mg / Ca",
      Na_Ca = "Na / Ca"
    )
  ) |>
  filter(is.finite(Ratio_Value), !is.na(Mountain_Side))

n_under <- ratio_df |>
